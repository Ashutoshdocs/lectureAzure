"""
Employee Lookup — SQL Injection / WAF Demo Lab
================================================
A deliberately vulnerable Flask app for demonstrating SQL injection and how a
Web Application Firewall (e.g. Azure Application Gateway WAF) blocks it.

  * /            -> nice UI + JSON/HTML results
  * /?id=1       -> normal lookup           (WAF: allowed)
  * /?id=1 OR 1=1-> injection, dumps table  (WAF: should block)
  * mode=secure  -> same query, parameterized (injection fails safely)
  * /health      -> "Application Healthy"   (for App Gateway health probe)

Run:  python3 app.py   ->  http://0.0.0.0:5000
"""

from flask import Flask, request, jsonify, render_template_string
import sqlite3
import os

DB = "employee.db"
app = Flask(__name__)


# ---------------------------------------------------------------------------
# Database bootstrap (so the lab is self-contained — no manual sqlite3 steps)
# ---------------------------------------------------------------------------
def init_db():
    if os.path.exists(DB):
        return
    conn = sqlite3.connect(DB)
    cur = conn.cursor()
    cur.execute("CREATE TABLE employees (id INTEGER, name TEXT, role TEXT, salary INTEGER)")
    cur.executemany(
        "INSERT INTO employees VALUES (?,?,?,?)",
        [
            (1, "John",  "Engineer",       92000),
            (2, "Alice", "Security Lead",  138000),
            (3, "Bob",   "Analyst",        71000),
            (4, "David", "Admin",          64000),
        ],
    )
    conn.commit()
    conn.close()

# ---------------------------------------------------------------------------
# Routes
# ---------------------------------------------------------------------------
@app.route("/health")
def health():
    return "Application Healthy", 200


@app.route("/")
def home():
    empid = request.args.get("id")
    mode = request.args.get("mode", "vuln")

    # Landing page with no lookup yet
    if empid is None:
        return render_template_string(
            PAGE, empid=None, mode=mode, shown_query=None,
            rows=None, error=None, leaked=False,
        )

    conn = sqlite3.connect(DB)
    cur = conn.cursor()
    rows, error, shown_query = None, None, ""

    try:
        if mode == "secure":
            # THE FIX: bound parameter — input can never change the query structure
            shown_query = "SELECT * FROM employees WHERE id = ?     -- param: " + repr(empid)
            rows = cur.execute("SELECT * FROM employees WHERE id = ?", (empid,)).fetchall()
        else:
            # THE VULNERABILITY: user input concatenated straight into SQL
            query = f"SELECT * FROM employees WHERE id={empid}"
            shown_query = query
            rows = cur.execute(query).fetchall()
    except sqlite3.Error as e:
        error = str(e)
    finally:
        conn.close()

    # Detect the classic tell: a by-id lookup returned more than one row
    leaked = bool(mode != "secure" and rows and len(rows) > 1)

    # Content negotiation: browsers get the UI, curl/tools get JSON
    if "text/html" not in request.headers.get("Accept", "") or request.args.get("format") == "json":
        return jsonify(query=shown_query, mode=mode, rows=rows, error=error, leaked=leaked)

    return render_template_string(
        PAGE, empid=empid, mode=mode, shown_query=shown_query,
        rows=rows, error=error, leaked=leaked,
    )


if __name__ == "__main__":
