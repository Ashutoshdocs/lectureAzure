# ==========================================================
# STEP 1 - Create Resource Group
# Purpose: Container for all Azure resources
# ==========================================================
az group create --name datafactoryblaze --location centralindia

# ==========================================================
# STEP 2 - Create Storage Account
# Purpose: Store CSV file
# ==========================================================
az storage account create --resource-group datafactoryblaze --name adfstorageblaze --location centralindia --sku Standard_LRS

# ==========================================================
# STEP 3 - Create Blob Container
# Purpose: Folder inside Storage Account
# ==========================================================
az storage container create --account-name adfstorageblaze  --name inputdata --auth-mode login

# ==========================================================
# STEP 4 - Upload CSV File
# Purpose: Upload source data
# ==========================================================


# ==========================================================
# STEP 5 - Create Azure SQL Server
# Purpose: Host SQL databases
# ==========================================================
az sql server create --resource-group datafactoryblaze --name adfsqlserverblaze --location centralindia --admin-user azureadmin --admin-password Password@12345

# ==========================================================
# STEP 6 - Create Azure SQL Database
# Purpose: Destination database
# ==========================================================
az sql db create --resource-group datafactoryblaze --server adfsqlserverblaze --name salesdb --service-objective Basic

# ==========================================================
# STEP 7 - Allow Azure Services to Connect
# Purpose: Allow ADF to access SQL Database
# ==========================================================
az sql server firewall-rule create --resource-group datafactoryblaze --server adfsqlserverblaze --name AllowAzure --start-ip-address 0.0.0.0 --end-ip-address 0.0.0.0

# ==========================================================
# STEP 8 - Create Azure Data Factory
# Purpose: Build ETL Pipelines
# ==========================================================
az datafactory create --resource-group datafactoryblaze --name adfblazefactory --location centralindia


-- Drop tables if they already exist
IF OBJECT_ID('dbo.SalesGST', 'U') IS NOT NULL
    DROP TABLE dbo.SalesGST;

IF OBJECT_ID('dbo.Sales', 'U') IS NOT NULL
    DROP TABLE dbo.Sales;

-- Source table (for Copy Activity if needed)
CREATE TABLE Sales
(
    Id INT PRIMARY KEY,
    Name VARCHAR(100),
    City VARCHAR(100),
    Product VARCHAR(100),
    Quantity INT,
    Amount DECIMAL(10,2)
);

-- Destination table (for Mapping Data Flow)
CREATE TABLE SalesGST
(
    Id INT PRIMARY KEY,
    Name VARCHAR(100),
    City VARCHAR(100),
    Product VARCHAR(100),
    Quantity INT,
    Amount DECIMAL(10,2),
    GST DECIMAL(10,2)
);

-- Verify tables
SELECT * FROM Sales;
SELECT * FROM SalesGST;
